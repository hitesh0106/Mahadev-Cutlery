   495→   495→/**
   496→   496→ * Resolves a storefront label (chip, nav link, search hit) to a browse target.
   497→   497→ * Order: exact department → exact brand (also "Brand · qualifier" chips) →
   498→   498→ * product-type alias. Returns null when nothing approved matches (callers then
   499→   499→ * keep the element non-navigating instead of inventing a destination).
   500→   500→ */
   501→   501→export function resolveChipTarget(raw: string): ChipTarget | null {
   502→   502→  const label = raw.trim();
   503→   503→  if (!label) return null;
   504→   504→
   505→   505→  if (DEPARTMENTS_EIGHT.some((d) => d.name === label)) {
   506→   506→    return { kind: 'department', name: label };
   507→   507→  }
   508→   508→
   509→   509→  const base = label.split('·')[0].trim();
   510→   510→  if (BRAND_LOOKUP.has(base)) return { kind: 'brand', name: base };
   511→   511→  if (BRAND_LOOKUP.has(label)) return { kind: 'brand', name: label };
   512→   512→
   513→   513→  const hit = typeEntry(label);
   514→   514→  if (hit) return { kind: 'type', name: label };
   515→   515→
   516→   516→  return null;
   517→   517→}
   518→   518→
   519→   519→/** Type-entry lookup used by the type collection view (exact, then keyword). */
   520→   520→export function typeEntry(name: string): { depts: string[]; kw: string[] } | null {
   521→   521→  const l = name.trim().toLowerCase();
   522→   522→  if (TYPE_INDEX[l]) return TYPE_INDEX[l];
   523→   523→  for (const entry of Object.values(TYPE_INDEX)) {
   524→   524→    if (entry.kw.some((k) => l.includes(k) || k.includes(l))) return entry;
   525→   525→  }
   526→   526→  return null;
   527→   527→}

   529→   529→/** Strips "(…)" qualifiers from a brand's primaryDepartments entry. */
   530→   530→export function baseDepartment(entry: string): string | null {
   531→   531→  const base = entry.replace(/\s*\(.*\)\s*$/, '').trim();
   532→   532→  if (base === 'Per actual verified product') return null;
   533→   533→  return DEPARTMENTS_EIGHT.some((d) => d.name === base) ? base : null;
   534→   534→}

   536→   536→/** Approved departments for a brand (qualified entries resolved). */
   537→   537→export function brandDepartments(brand: string): string[] {
   538→   538→  const profile = BRANDS.find((b) => b.brand === brand);
   539→   539→  let sources: string[] = [];
   540→   540→  if (profile) {
   541→   541→    sources = profile.primaryDepartments;
   542→   542→  } else {
   543→   543→    // Brands without a full profile (e.g. O3+) fall back to their group.
   544→   544→    const group = BRAND_GROUPS.find((g) => (g.brands as readonly string[]).includes(brand))?.group;
   545→   545→    sources = group ? (GROUP_DEPTS[group] ?? []) : [];
   546→   546→  }
   547→   547→  const out: string[] = [];
   548→   548→  for (const entry of sources) {
   549→   549→    const dept = baseDepartment(entry);
   550→   550→    if (dept && !out.includes(dept)) out.push(dept);
   551→   551→  }
   552→   552→  return out;
   553→   553→}
