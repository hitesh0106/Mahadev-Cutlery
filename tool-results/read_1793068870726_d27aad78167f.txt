     1→'use client';

     3→// Preview i18n — EN / हिन्दी / ગુજરાતી interface language for the showcase app.
     4→// Mirrors the Shopify theme's bundled locale files (en.default.json / hi.json /
     5→// gu.json): interface strings only. Brand names, product titles, prices and the
     6→// WhatsApp order message (store-facing) are never translated.
     7→//
     8→// Implementation: external store + useSyncExternalStore (same SSR-safe pattern
     9→// as the wishlist), persisted to localStorage 'mahadev-preview:lang'.
    10→
    11→import { useSyncExternalStore } from 'react';
    12→
    13→export type Lang = 'en' | 'hi' | 'gu';
    14→
    15→export const LANGUAGES: { code: Lang; label: string; short: string; native: boolean }[] = [
    16→  { code: 'en', label: 'English', short: 'EN', native: true },
    17→  { code: 'hi', label: 'हिन्दी', short: 'हिं', native: true },
    18→  { code: 'gu', label: 'ગુજરાતી', short: 'ગુ', native: true },
    19→];
    20→
    21→const LANG_KEY = 'mahadev-preview:lang';
    22→const LANG_EVENT = 'mahadev-preview:lang-changed';
    23→
    24→type Listener = () => void;
    25→const listeners = new Set<Listener>();
    26→let langCache: Lang | null = null;
    27→
    28→function readLang(): Lang {
    29→  try {
    30→    const stored = localStorage.getItem(LANG_KEY);
    31→    return stored === 'hi' || stored === 'gu' ? stored : 'en';
    32→  } catch {
    33→    return 'en';
    34→  }
    35→}
    36→
    37→function getLangSnapshot(): Lang {
    38→  if (langCache === null) langCache = readLang();
    39→  return langCache;
    40→}
    41→
    42→function getServerLang(): Lang {
    43→  return 'en';
    44→}
    45→
    46→function subscribeLang(listener: Listener) {
    47→  listeners.add(listener);
    48→  const onChanged = () => listener();
    49→  window.addEventListener(LANG_EVENT, onChanged);
    50→  window.addEventListener('storage', onChanged);
    51→  return () => {
    52→    listeners.delete(listener);
    53→    window.removeEventListener(LANG_EVENT, onChanged);
    54→    window.removeEventListener('storage', onChanged);
    55→  };
    56→}
    57→
    58→export function setLang(next: Lang) {
    59→  langCache = next;
    60→  try {
    61→    localStorage.setItem(LANG_KEY, next);
    62→  } catch {
    63→    // Storage unavailable — language stays for this session only.
    64→  }
    65→  listeners.forEach((listener) => listener());
    66→  window.dispatchEvent(new Event(LANG_EVENT));
    67→  document.documentElement.lang = next === 'en' ? 'en' : next;
    68→}
    69→
    70→/** Keep <html lang> in sync (called by the language switcher). */
    71→export function syncDocumentLang() {
    72→  document.documentElement.lang = getLangSnapshot() === 'en' ? 'en' : getLangSnapshot();
    73→}

   739→
   740→export function translate(lang: Lang, text: string): string {
   741→  if (lang === 'en') return text;
   742→  return DICTS[lang][text] ?? text;
   743→}
   744→
   745→export function translateFormat(
   746→  lang: Lang,
   747→  pattern: string,
   748→  vars: Record<string, string | number>,
   749→): string {
   750→  const translated = translate(lang, pattern);
   751→  return translated.replace(/\{(\w+)\}/g, (match, key: string) =>
   752→    key in vars ? String(vars[key]) : match,
   753→  );
   754→}

   762→      translateFormat(lang, pattern, vars),
   763→  };
   764→}
